اور داؤد نے کہا، میں نے اب کیا کیا ہے؟ کیا کوئی سبب نہیں؟ –1 سموئیل 17:29۔

داؤد اور جالوت کی کہانی تین ہزار سال سے بھی زیادہ پہلے پیش آئی، اور تب سے جوانوں اور بوڑھوں میں ہمیشہ سے مقبول رہی ہے۔ کمزور فریق کی حالت میں کچھ ایسا ہے جس سے ہم سب اپنا تعلق محسوس کرتے ہیں۔ شاید لاشعوری طور پر ہم ایسے حالات میں خود کو دیکھتے ہیں اور دل ہی دل میں امید کرتے ہیں کہ ایک دن ہماری اپنی قسمت بھی اسی طرح چمک اٹھے گی۔ افسوس کہ اس قسم کی اتفاقی کامیابی حقیقی زندگی سے زیادہ ہالی وڈ کی کسی فلم جیسی لگتی ہے۔ ہماری نظر صرف جوان اور بوڑھے، چرواہے اور جنگجو، اور بھاری جنگی ہتھیاروں اور فلاخن کے درمیان تصادم پر جاتی ہے۔ لیکن کچھ ابتدائی اصول تھے جنہیں داؤد نے اپنی جوانی میں اپنایا، اور انہی نے اسے کامیابی کے لیے موزوں امیدوار بنایا۔ مگر فیصلہ کن رومانوی فتح سے ہماری محبت میں اکثر ان اصولوں کا ذکر نہیں ہوتا۔

اگر اپنے بارے میں ہمارا واحد تصور ہمارے اردگرد کے سماجی آئینے سے آتا ہے تو ہمارے اندر اپنی ایک بگڑی ہوئی تصویر بن جائے گی، بالکل ویسی جیسی کسی میلے کے ٹیڑھے آئینے میں جھانکنے سے دکھائی دیتی ہے۔ ہمارے ماحول کی سماجی رائے اور تصورات اکثر ہماری آنکھوں اور کانوں کے سامنے ہماری شخصیت کی ایک بے ڈھنگی اور مسخ شدہ تصویر پیش کرتے ہیں۔ بعض اوقات یہ غیر حقیقی طور پر مثبت ہوتی ہے اور ذرائع ابلاغ کی نظر میں کسی شخص کو اس وقت بھی بڑھا چڑھا کر پیش کرتی ہے جب اس میں بہادری کی صفات بہت کم دکھائی دیتی ہیں۔

کھیلوں کا میدان شاید اس معاملے میں سب سے بڑا قصوروار ہے۔ ان لوگوں کی تعریف کی جاتی ہے جو دوسروں سے بہتر دوڑ سکتے، پھینک سکتے یا نشانہ لگا سکتے ہیں۔ لیکن کھیلوں کا ہنر انسان کی حیثیت سے ان کی حقیقی اور خالص قدر میں بہت کم اضافہ کرتا ہے۔ سکے کا دوسرا رخ ایسی بہت سی منفی اور تحقیر آمیز باتیں سامنے لاتا ہے جو کوشش کرنے کی ہر سعی کو ناکام بنانے کے درپے ہوتی ہیں۔ یہ باتیں دباؤ میں مبتلا والدین، عدم تحفظ کے شکار بہن بھائیوں، مایوس مالکان اور شناخت کے لیے جدوجہد کرنے والے “سب سے موزوں ہی کی بقا” والے معاشرے کے بے شمار دوسرے جاننے والوں کی طرف سے موسلادھار برستی ہیں۔ ان دونوں گروہوں کی باتیں بے ربط بھی ہیں اور غیر متناسب بھی۔ آپ کو عظیم کہا جانا آپ کو کبھی عظیم نہیں بنائے گا، بالکل اسی طرح جیسے تحقیر آمیز باتوں کی بوچھاڑ آپ میں فروتنی پیدا نہیں کرے گی۔ افسوس کہ ہم میں سے بہت سے لوگ اکثر ان غیر متناسب عکس بندیوں کو اپنی سوچ پر اثر انداز ہونے اور اپنے راستے کا تعین کرنے دیتے ہیں۔

باقی مخلوقات کے مقابلے میں انسان کو دیا گیا سب سے بڑا اور اعلیٰ عطیہ اس کی جسمانی صلاحیتوں میں نہیں بلکہ اس کے ذہن میں ہے۔ سب سے ذہین جانور کو ذمہ دار بننے کی تربیت دی جا سکتی ہے، لیکن وہ اس تربیت کا انتخاب یا اس کی سمت متعین نہیں کر سکتا۔ مگر انسان ہونے کے ناتے ہم اپنے لیے بالکل نئے پروگرام لکھ سکتے ہیں، جو ہماری جبلتوں یا ماحول کی تربیت سے پوری طرح آزاد ہوں۔ ہم میں سے جو اپنی جبلتوں اور تربیت کے مطابق فطری طور پر زندگی گزارتے ہیں، وہ شاید ہی حیوانی دنیا سے بلند ہو پائیں۔ لیکن جو ان عطیات کو بروئے کار لائیں گے اور ترقی دیں گے، وہ دیوؤں سے اپنے مقابلے میں منفرد طور پر وہ کردار ادا کریں گے جو خدا نے ان کے لیے مقرر کیا ہے۔ داؤد کا زمانہ اور مقام منفرد تھے، مگر اس کے اپنائے ہوئے اصول آفاقی تھے۔ اگر میں اپنے کتے کو یودل گانا سکھا دوں تو اس سے ثابت ہوگا کہ کتے یودل گا سکتے ہیں، اور دوسروں کے لیے بھی ایسا کرنے کا امکان پیدا ہوگا۔ اسی طرح، اگر داؤد نے صرف معجزے کے ذریعے جالوت کو شکست دی تھی، تو ہمارے لیے کوئی امید نہیں سوائے اس کے کہ بیٹھ کر انتظار کریں کہ خدا ہمارے لیے بھی ایسا ہی کرے۔ لیکن اگر فتح کی بنیاد اصول تھے تو ہمارے پاس پیروی کے لیے ایک مثال اور اپنے میدانِ جنگ میں کام آنے والی ایک خوبی حاصل کرنے کا موقع ہے۔

جب داؤد اس صبح سویرے اٹھا اور اپنے بھائیوں کے لیے مکئی، روٹی اور پنیر کی گٹھڑی کندھے پر رکھی تو اسے اندازہ نہ تھا کہ وہ دن کیا لے کر آئے گا۔ خداوند اس کا چرواہا تھا، اسے کوئی کمی نہ تھی، لیکن جب وہ اس منظر پر پہنچا تو یہ دیکھ کر حیران رہ گیا۔ دونوں لشکر آمنے سامنے کے پہاڑوں پر صف آرا تھے اور تعطل کا شکار تھے۔ اپنا پورا لشکر وادی میں اتار کر مخالفین سے چڑھائی پر لڑنا آپ کو بہت بڑے نقصان میں ڈال دیتا۔ کوئی بھی فریق یہ خطرہ مول لینے کو تیار نہ تھا۔ چنانچہ فلستیوں نے جنگ کا ایک عام طریقہ اپناتے ہوئے تجویز پیش کی کہ دونوں طرف سے ایک ایک جنگجو وادی میں لڑے، تاکہ کسی کو فائدہ نہ ہو۔ لیکن فلستیوں کے لشکر میں ایک دیو تھا جس نے اسرائیل کے لشکر کو دہشت زدہ کر رکھا تھا۔ جب داؤد نے پوچھا کہ کوئی اس کافر کی ملامت آمیز کفر بکنے والی باتوں کا مقابلہ کیوں نہیں کرتا، تو اسے اپنے سب سے بڑے بھائی کی سخت سرزنش کا سامنا کرنا پڑا: …اور الیاب کا غصہ داؤد پر بھڑکا، اور اس نے کہا، تُو یہاں کیوں آیا ہے؟ اور بیابان میں اُن چند بھیڑوں کو کس کے پاس چھوڑ آیا ہے؟ میں تیرے غرور اور تیرے دل کی شرارت کو جانتا ہوں، کیونکہ تُو لڑائی دیکھنے کے لیے آیا ہے۔ داؤد نے کہا، میں نے اب کیا کیا ہے؟ کیا کوئی سبب نہیں؟ –1 سموئیل 17:28-29۔

اپنے ایمان کے بانی اور کامل کرنے والے یسوع کو تکتے رہیں؛ جس نے اُس خوشی کے لیے جو اُس کے سامنے دھری تھی صلیب کا دکھ سہا اور شرمندگی کی پروا نہ کی، اور خدا کے تخت کی دہنی طرف بیٹھ گیا ہے –عبرانیوں 12:2۔ جب کسی شخص کے سامنے کوئی مقصد ہو تو وہ تقریباً ہر چیز برداشت کر سکتا ہے۔ بعض لوگ فانی تاج کے لیے برداشت کرتے ہیں۔ بادشاہ ساؤل نے ٹیکس سے مستثنیٰ دولت اور داماد بننے کا مرتبہ پیش کیا تھا، لیکن دنیاوی جہیز کے لالچ میں کوئی رضاکار نہ بنا۔ زمینی عظمت کا خیال نرم دل داؤد کے لیے بے معنی تھا، لیکن جالوت کی کفر بکنے والی طعنہ زنی نے اس کے اندر غیرت کی آگ بھڑکا دی اور وہ ایمان سے آگے بڑھا۔ مگر ایمان محض کوئی وقتی خیال نہیں جسے جذبات کا ہوا بھرنے والا پمپ پھلا دے۔ کیونکہ جس طرح جسم روح کے بغیر مردہ ہے، اسی طرح ایمان بھی اعمال [عمل] کے بغیر مردہ ہے –یعقوب 2:26۔ یوں مقصد کو صاف طور پر سامنے رکھتے ہوئے اور اپنے خدا کو اپنے ساتھ پا کر، داؤد جالوت سے لڑنے کی پیشکش کرتا ہے۔

پس داؤد نے آگے بڑھ کر خدا کے مطابق، غیر مسخ شدہ اصولوں پر عمل کیا…

دردمند ضمیر انسان کے نفس میں بسا ہوتا ہے اور طے کرتا ہے کہ ہمیں کس چیز کی پروا ہے۔ بہت سے لوگ سمجھتے ہیں کہ ضمیر صرف روکنے والی رہنمائی کے طور پر کارآمد ہے۔ لیکن انسانی ضمیر کو اس کے برعکس کام کرنے کے لیے بھی پیدا کیا گیا تھا۔ جب ہمیں یقین ہو جائے کہ کوئی چیز اچھی اور درست ہے تو ہم جوش و خروش سے اس کے پیچھے چلتے ہیں۔ تھامس ایڈیسن نے کہا تھا کہ ذہانت ایک فیصد الہام اور ننانوے فیصد پسینہ بہانے کا نام ہے۔ یہ بات کافی حد تک درست ہو سکتی ہے، مگر اس میں کوئی شک نہیں کہ الہام پہلے آنا چاہیے۔ راخل کے لیے یعقوب کی آرزو نے اسے سات سال کی محنت بھلا دی۔ کامیاب بلب بنانے کے لیے ایڈیسن کی جستجو نے اسے ناکام ہونے والے ہزاروں اجزا بھلا دیے۔

بے پروائی اور بے حسی اس کا الٹ ہے اور یہ ضمیر کے بے حس ہو جانے کا نتیجہ ہے۔ داؤد حیران تھا کہ جالوت کی کفر بکنے والی طعنہ زنی سے خدا کے لشکر میں کسی کے اندر عمل کرنے کا کوئی سبب پیدا نہیں ہوا۔ کسی شخص کو جینے کی وجہ دے دو اور وہ طریقہ خود ڈھونڈ لے گا –نِتسکے۔ نازیوں کے حراستی کیمپ میں ہولوکاسٹ کے قیدی کی حیثیت سے وکٹر فرانکل کو کیمپ کے اس وارڈ میں کام سونپا گیا جہاں قیدی ٹائیفس سے مر رہے تھے۔ اس نے دیکھا کہ جب کوئی شخص امید چھوڑ دیتا تو جلد ہی موت اس کے پیچھے آتی۔ اسی طرح آج ہم میں سے بہت سوں کے لیے، جب کشمکش میں ذاتی فائدے کی امید ختم ہو جاتی ہے تو اس پر غالب آنے کا جوش بھی ختم ہو جاتا ہے۔ آج اکثر کارکن اچھا اور قابلِ قدر کام کرنے سے زیادہ اپنی تنخواہ کی فکر کرتے ہیں۔ اکثر جب کسی مرد کے دل میں اپنی بیوی کے لیے محبت کے جذبات نہیں رہتے تو وہ سمجھتا ہے کہ وہ اس سے محبت کرنے کے قابل نہیں؛ لیکن خدا کے اصولوں کے لیے صحت مند ضمیر رکھنے والا مرد اپنے جذبات یا بیوی کے ردعمل سے قطع نظر، اس کے ساتھ محبت بھرا برتاؤ کرے گا۔ ضمیر مقصد کو ابھارتا اور اس میں ایندھن بھرتا ہے، اور پُرجوش عمل کی بنیاد رکھتا ہے۔

دیانت دار تخیل روح کے دائرے سے تعلق رکھتا ہے اور ہمارے ضمیر سے تحریک پاتا ہے۔ ضرورت ایجاد کی ماں ہے –نامعلوم۔ کچھ لوگ زندگی کے دریافت ہونے کے منتظر ہیں، جب کہ دوسرے اسے چلتے چلتے تخلیق کر رہے ہیں۔ یہ جھوٹے آدرشوں کا دعویٰ کرنے کا اجازت نامہ نہیں۔ داؤد کی جستجو دیانت دار ہے، کیونکہ اس کے اردگرد کے ذرائع ابلاغ نے اس کی سوچ کو مسخ نہیں کیا تھا۔ بیت لحم کی پہاڑیوں میں وہ درندوں (ایک شیر اور ایک ریچھ) پر کامیابی حاصل کر چکا تھا، اور خدا کی مدد سے وادیِ ایلہ میں ایک ایسے شخص سے مقابلہ کرنے سے نہیں ڈرتا جو اپنی جوانی ہی سے جنگجو تھا۔
